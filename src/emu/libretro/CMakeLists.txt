# The libretro frontend: a fourth one beside qt, android and ios.
#
# It is a shared object the frontend loads, so everything it links has to be
# position independent - CMAKE_POSITION_INDEPENDENT_CODE is forced ON for the
# whole tree when this is enabled (see the root CMakeLists), since the emulator
# libraries are static and built for an executable otherwise.

# The files the emulator reads at run time, which standalone keeps beside its
# executable: the shaders of the graphics backend, the MIDI banks and the
# patch libraries. A core is one file, so they are compiled into it and
# written out under the frontend's system directory at start (state.cpp).
set(EKA2L1_EMBEDDED_DATA "${CMAKE_CURRENT_BINARY_DIR}/embedded_data.cpp")
set(EKA2L1_EMBED_NAMES "")
set(EKA2L1_EMBED_FILES "")
if (ANDROID)
    file(GLOB EKA2L1_SHADERS "${PROJECT_SOURCE_DIR}/src/emu/drivers/resources/gles/*")
else()
    file(GLOB EKA2L1_SHADERS "${PROJECT_SOURCE_DIR}/src/emu/drivers/resources/gl/*")
endif()
file(GLOB EKA2L1_UPSCALE "${PROJECT_SOURCE_DIR}/src/emu/drivers/resources/upscale/*")
file(GLOB EKA2L1_BANKS "${PROJECT_SOURCE_DIR}/src/emu/drivers/resources/defaultbank.*")
file(GLOB EKA2L1_PATCHES
    "${PROJECT_SOURCE_DIR}/src/patch/*/group/*.dll"
    "${PROJECT_SOURCE_DIR}/src/patch/*/group/*.map")
foreach(GROUP_AND_DIR "EKA2L1_SHADERS|resources" "EKA2L1_UPSCALE|resources/upscale" "EKA2L1_BANKS|resources" "EKA2L1_PATCHES|patch")
    string(REPLACE "|" ";" GROUP_AND_DIR "${GROUP_AND_DIR}")
    list(GET GROUP_AND_DIR 0 GROUP)
    list(GET GROUP_AND_DIR 1 DIR)
    foreach(DATA_FILE IN LISTS ${GROUP})
        if (IS_DIRECTORY "${DATA_FILE}")
            continue()
        endif()
        get_filename_component(DATA_NAME "${DATA_FILE}" NAME)
        list(APPEND EKA2L1_EMBED_NAMES "${DIR}/${DATA_NAME}")
        list(APPEND EKA2L1_EMBED_FILES "${DATA_FILE}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DATA_FILE}")
    endforeach()
endforeach()

set(EKA2L1_EMBED_SOURCE "// Generated by src/emu/libretro/CMakeLists.txt\n#include <cstddef>\nnamespace eka2l1::libretro {\nstruct embedded_file { const char *path; const unsigned char *data; std::size_t size; };\n")
set(EKA2L1_EMBED_TABLE "")
list(LENGTH EKA2L1_EMBED_FILES EKA2L1_EMBED_COUNT)
set(EKA2L1_EMBED_INDEX 0)
while(EKA2L1_EMBED_INDEX LESS EKA2L1_EMBED_COUNT)
    list(GET EKA2L1_EMBED_FILES ${EKA2L1_EMBED_INDEX} DATA_FILE)
    list(GET EKA2L1_EMBED_NAMES ${EKA2L1_EMBED_INDEX} DATA_NAME)
    file(READ "${DATA_FILE}" DATA_HEX HEX)
    string(LENGTH "${DATA_HEX}" DATA_HEX_LENGTH)
    math(EXPR DATA_SIZE "${DATA_HEX_LENGTH} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," DATA_BYTES "${DATA_HEX}")
    string(APPEND EKA2L1_EMBED_SOURCE "static const unsigned char data_${EKA2L1_EMBED_INDEX}[] = {${DATA_BYTES}0};\n")
    string(APPEND EKA2L1_EMBED_TABLE "    { \"${DATA_NAME}\", data_${EKA2L1_EMBED_INDEX}, ${DATA_SIZE} },\n")
    math(EXPR EKA2L1_EMBED_INDEX "${EKA2L1_EMBED_INDEX} + 1")
endwhile()
string(APPEND EKA2L1_EMBED_SOURCE "extern const embedded_file embedded_files[] = {\n${EKA2L1_EMBED_TABLE}};\nextern const std::size_t embedded_file_count = ${EKA2L1_EMBED_COUNT};\n}\n")
file(WRITE "${EKA2L1_EMBEDDED_DATA}" "${EKA2L1_EMBED_SOURCE}")

add_library(eka2l1_libretro SHARED
    "${EKA2L1_EMBEDDED_DATA}"
    src/main.cpp
    src/content.cpp
    src/hooks.cpp
    src/state.cpp
    src/thread.cpp
    include/libretro.h
    include/libretro_state.h)

target_include_directories(eka2l1_libretro PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include")

# GIT_BRANCH / GIT_COMMIT_HASH and the rest come from the generated headers the
# emulator libraries already carry.

# The same set the Qt frontend links, minus Qt itself.
target_link_libraries(eka2l1_libretro PRIVATE
    common
    config
    drivers
    epoc
    epocservs
    epocutils
    epockern
    epocpkg
    gdbstub
    uvw
    j2me
    stb
    yaml-cpp)

# Libretro cores are named <core>_libretro.so, with no "lib" prefix.
# The version reported is upstream's, with the upstream commit this core is
# rebased on, from upstream.version
file(STRINGS "${PROJECT_SOURCE_DIR}/upstream.version" EKA2L1_UPSTREAM_COMMIT LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/upstream.version")
target_compile_definitions(eka2l1_libretro PRIVATE
    EKA2L1_UPSTREAM_COMMIT="${EKA2L1_UPSTREAM_COMMIT}")

# RetroArch's names: eka2l1_libretro_android.so on Android
if (ANDROID)
    set(EKA2L1_LIBRETRO_NAME eka2l1_libretro_android)
else()
    set(EKA2L1_LIBRETRO_NAME eka2l1_libretro)
endif()

set_target_properties(eka2l1_libretro PROPERTIES
    OUTPUT_NAME ${EKA2L1_LIBRETRO_NAME}
    PREFIX ""
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin"
    LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")

# On Linux and Android the core binds its own symbols inside itself: the
# frontend's libraries do not stand in for them, and FFmpeg's x86 assembly,
# which refers to its tables directly, links into a shared object
if (UNIX AND NOT APPLE)
    target_link_options(eka2l1_libretro PRIVATE -Wl,-Bsymbolic)
endif()

# The GL loader asks EGL for the functions on Android
if (ANDROID)
    target_link_libraries(eka2l1_libretro PRIVATE EGL GLESv3 log android)
endif()
